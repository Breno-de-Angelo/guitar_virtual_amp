\subsection{Commands}

\begin{verbatim}
GVA_TONE_MANIFEST=~/.cache/gva_test/prepared/eval_egdb.tsv \
  cargo test --release --workspace --exclude guitar_android eval_tone \
  -- --ignored --nocapture
\end{verbatim}

\begin{center}
\small
\begin{tabularx}{\textwidth}{@{}>{\raggedright}p{4.2cm}X@{}}
\toprule
Variable & Effect \\
\midrule
\code{GVA_TONE_LIMIT} & number of items (default 8; the recorded runs used 12) \\
\code{GVA_TONE_ITEMS=105,125} & only these items \\
\code{GVA_TONE_VERBOSE=1} & print each estimate's settings and match EQ \\
\code{GVA_TONE_GENS} & CMA-ES generations (default 30) \\
\code{GVA_TONE_SECS} & excerpt length in seconds (default 12) \\
\code{GVA_TONE_ORACLE=0} & skip the true-DI search \\
\code{GVA_TONE_OUT=<dir>} & write WAVs (target, estimate, preset) for listening \\
\code{GVA_TONE_STEMS=<tsv>} & separated-stem targets (\file{eval_mix_bsrofo_sw.tsv}) \\
\code{GVA_TONE_STEMS_ONLY=1} & only items that have a stem (the recorded runs) \\
\code{GVA_TONE_SAMPLES=<di:notes,...>} & sampler stand-in \\
\code{GVA_TONE_FLOOR} & search floor in dB \\
\code{GVA_TONE_PRIOR} & weight of the preset prior (default 1) \\
\code{GVA_TONE_PLUCK=0} & skip the pluck stand-in \\
\code{GVA_TONE_COLOUR=1} & search the guitar colour with the chain \\
\code{GVA_TONE_ABLATE=1} & oracle stand-ins with the DI's colour, dynamics, both \\
\code{GVA_TONE_CALIBRATE=1} & stand-in colour calibrated on other recordings \\
\code{GVA_TONE_CALIBRATION_ITEMS} & how many (default 3) \\
\bottomrule
\end{tabularx}
\end{center}

\code{pluck_vs_di} and \code{fit_pluck} take
\cmd{GVA_PLUCK_ITEMS=<di.wav>:<notes.txt>,...} (Guitar-TECHS DI only, never EGDB).
